```sql
    created_at

-- All should have:
-- - recommendation_stage = 'cold_start'
-- - interaction_count = 0
```

#### Join Query - Complete User View
```sql
SELECT 
    u.user_id,
    u.email,
    u.name,
    p.research_stage,
    p.primary_domain,
    p.reading_level,
    p.profile_completeness,
    urs.recommendation_stage,
    urs.interaction_count,
    COUNT(uih.interest_term) as total_interests
FROM users u
LEFT JOIN user_profiles_extended p ON u.user_id = p.user_id
LEFT JOIN user_recommendation_state urs ON u.user_id = urs.user_id
LEFT JOIN user_interest_hierarchy uih ON u.user_id = uih.user_id
GROUP BY 
    u.user_id, u.email, u.name, 
    p.research_stage, p.primary_domain, p.reading_level, p.profile_completeness,
    urs.recommendation_stage, urs.interaction_count
ORDER BY u.user_id;

-- Verify all relationships are correct
```

### Test 3.2: Verify Using Python Script

```bash
# Run database verification
docker-compose exec api python -c "
import asyncio
from app.db.connection import db

async def verify():
    await db.connect()
    
    print('\n' + '='*60)
    print('DATABASE VERIFICATION')
    print('='*60)
    
    # Check users
    users = await db.fetch('SELECT COUNT(*) as count FROM users')
    print(f'\n📊 Users: {users[0][\"count\"]}')
    
    # Check profiles
    profiles = await db.fetch('SELECT COUNT(*) as count FROM user_profiles_extended')
    print(f'📊 Profiles: {profiles[0][\"count\"]}')
    
    # Check interests
    interests = await db.fetch('SELECT COUNT(*) as count FROM user_interest_hierarchy')
    print(f'📊 Total Interests: {interests[0][\"count\"]}')
    
    # Check states
    states = await db.fetch('SELECT COUNT(*) as count FROM user_recommendation_state')
    print(f'📊 States: {states[0][\"count\"]}')
    
    # Check completeness distribution
    completeness = await db.fetch('''
        SELECT 
            AVG(profile_completeness) as avg,
            MIN(profile_completeness) as min,
            MAX(profile_completeness) as max
        FROM user_profiles_extended
    ''')
    
    if completeness[0]['avg']:
        print(f'\n📈 Profile Completeness:')
        print(f'   Average: {completeness[0][\"avg\"]:.2f}')
        print(f'   Min: {completeness[0][\"min\"]:.2f}')
        print(f'   Max: {completeness[0][\"max\"]:.2f}')
    
    print('\n' + '='*60)
    
    await db.disconnect()

asyncio.run(verify())
"
```

**Success Criteria:**
- ✅ User count matches registrations
- ✅ Profile count matches user count
- ✅ Interest count >= 3 × user count
- ✅ State count matches user count
- ✅ Average completeness > 0.5

## 📝 Phase 7: Automated Test Script (Test NOW)

Create this comprehensive test script:

```bash
# Save as: test_complete_flow.sh
#!/bin/bash

set -e  # Exit on error

echo "🧪 CiteConnect Complete Testing Suite"
echo "======================================"

BASE_URL="http://localhost:8000/api/v1"
HEALTH_URL="http://localhost:8000/health"

# Colors
GREEN='\033[0;32m'
RED='\033[0;31m'
NC='\033[0m'

test_health() {
    echo -e "\n${GREEN}Test 1: Health Check${NC}"
    HEALTH=$(curl -s $HEALTH_URL)
    STATUS=$(echo $HEALTH | jq -r '.status')
    
    if [ "$STATUS" = "healthy" ]; then
        echo "✅ Health check passed"
    else
        echo "❌ Health check failed"
        exit 1
    fi
}

test_registration() {
    echo -e "\n${GREEN}Test 2: User Registration${NC}"
    
    RESPONSE=$(curl -s -X POST $BASE_URL/users/register \
      -H "Content-Type: application/json" \
      -d '{
        "email": "autotest@example.com",
        "password": "AutoTest123!",
        "full_name": "Auto Test User"
      }')
    
    USER_ID=$(echo $RESPONSE | jq -r '.user_id')
    TOKEN=$(echo $RESPONSE | jq -r '.access_token')
    
    if [ "$USER_ID" != "null" ] && [ "$TOKEN" != "null" ]; then
        echo "✅ Registration successful - User ID: $USER_ID"
        echo "$USER_ID" > /tmp/test_user_id
        echo "$TOKEN" > /tmp/test_token
    else
        echo "❌ Registration failed"
        echo $RESPONSE | jq '.'
        exit 1
    fi
}

test_profile_creation() {
    echo -e "\n${GREEN}Test 3: Profile Creation${NC}"
    
    USER_ID=$(cat /tmp/test_user_id)
    
    PROFILE_RESPONSE=$(curl -s -X POST $BASE_URL/users/$USER_ID/profile \
      -H "Content-Type: application/json" \
      -d '{
        "research_stage": "phd",
        "primary_domain": "healthcare",
        "reading_level": "advanced",
        "interests": ["machine learning", "medical imaging", "diagnostics"],
        "sub_domains": ["radiology"],
        "research_goals": ["publish_paper"],
        "years_experience": 3
      }')
```
